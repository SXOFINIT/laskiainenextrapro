WITH PlannedData AS ( /* ============================================================ 1. Suunnitelma: reseptirivit ============================================================ */ SELECT d.KST_ID AS COSTCENTERID, kst.KST_NAME AS COSTCENTERNAME, CAST(h.MPP_DATE AS date) AS MENUPLANDATE, d.SPT_NR AS MENUPLANTEMPLATEID, meal.MTI_NAME AS MENUPLANTEMPLATENAME, h.ART_NR AS SETMENUID, e.ART_NAME AS SETMENUNAME, h.MPP_WES AS COS_MENU, sp.SPA_NAME AS PLAN_GROUP, a.REZ_ID AS ITEMID, b.REZ_NAME AS ITEMNAME, 'RECIPE' AS ITEMTYPE, h.MPP_COVERS AS ORIGINAL_PLANNEDPORTIONS, a.MPP_FACTOR AS PORTIONFACTOR, a.MPP_PORTION AS STANDARDPORTION, a.MPP_GEW AS ORIGINAL_PLANNEDWEIGHT, a.MPP_WES AS COS_ITEM FROM dbo.REZ2MENU AS a INNER JOIN dbo.MENUPLANPOS AS h ON h.MPL_ID = a.MPL_ID AND h.MPP_POS = a.MPP_POS INNER JOIN dbo.MENUPLAN AS d ON d.MPL_ID = a.MPL_ID INNER JOIN dbo.BESTELLER AS f ON f.BST_ID = h.CHG_USER INNER JOIN dbo.REZEPTUR AS b ON b.REZ_ID = a.REZ_ID LEFT JOIN dbo.SPARTE AS sp ON sp.SPA_ID = b.SPA_ID LEFT JOIN dbo.ARTIKEL AS e ON e.ART_ID = h.ART_NR INNER JOIN dbo.KOSTST AS kst ON kst.KST_ID = d.KST_ID AND kst.AKTIV = ' ' INNER JOIN dbo.MYGSPLTEMPLATE AS tp ON tp.SPT_ID = d.SPT_NR AND tp.AKTIV = ' ' INNER JOIN dbo.MYGMEALTIME AS meal ON meal.MTI_ID = tp.MTI_ID WHERE b.REZ_NAME IS NOT NULL AND !KST! AND CAST(h.MPP_DATE AS date) >= :dfVon AND CAST(h.MPP_DATE AS date) <= :dfBis AND DATEDIFF(YEAR, :dfVon, :dfBis) <> 40 UNION ALL /* ============================================================ 2. Suunnitelma: varasto-/artikkelirivit ============================================================ */ SELECT d.KST_ID AS COSTCENTERID, kst.KST_NAME AS COSTCENTERNAME, CAST(h.MPP_DATE AS date) AS MENUPLANDATE, d.SPT_NR AS MENUPLANTEMPLATEID, meal.MTI_NAME AS MENUPLANTEMPLATENAME, h.ART_NR AS SETMENUID, e.ART_NAME AS SETMENUNAME, h.MPP_WES AS COS_MENU, sp.WGR_NAME AS PLAN_GROUP, a.ART_NR AS ITEMID, b.ART_NAME AS ITEMNAME, 'INVENTORY_ITEM' AS ITEMTYPE, h.MPP_COVERS AS ORIGINAL_PLANNEDPORTIONS, a.MPP_FACTOR AS PORTIONFACTOR, a.MPP_PORTION AS STANDARDPORTION, a.MPP_GEW AS ORIGINAL_PLANNEDWEIGHT, a.MPP_WES AS COS_ITEM FROM dbo.REZ2MENU AS a INNER JOIN dbo.MENUPLANPOS AS h ON h.MPL_ID = a.MPL_ID AND h.MPP_POS = a.MPP_POS INNER JOIN dbo.MENUPLAN AS d ON d.MPL_ID = a.MPL_ID INNER JOIN dbo.BESTELLER AS f ON f.BST_ID = h.CHG_USER INNER JOIN dbo.ARTIKEL AS b ON b.ART_ID = a.ART_NR LEFT JOIN dbo.WARENGRUPPE AS sp ON sp.WGR_ID = b.WGR_ID LEFT JOIN dbo.ARTIKEL AS e ON e.ART_ID = h.ART_NR INNER JOIN dbo.KOSTST AS kst ON kst.KST_ID = d.KST_ID AND kst.AKTIV = ' ' INNER JOIN dbo.MYGSPLTEMPLATE AS tp ON tp.SPT_ID = d.SPT_NR AND tp.AKTIV = ' ' INNER JOIN dbo.MYGMEALTIME AS meal ON meal.MTI_ID = tp.MTI_ID WHERE b.ART_NAME IS NOT NULL AND !KST! AND CAST(h.MPP_DATE AS date) >= :dfVon AND CAST(h.MPP_DATE AS date) <= :dfBis AND DATEDIFF(YEAR, :dfVon, :dfBis) <> 40 ), ActualData AS ( /* ============================================================ 3. Toteumarivit ============================================================ */ SELECT p.COSTCENTERID, p.COSTCENTERNAME, CAST(p.MENUPLANDATE AS date) AS MENUPLANDATE, p.MENUPLANTEMPLATEID, p.MENUPLANTEMPLATENAME, p.SETMENUID, g.SMG_NAME AS SET_MENU_GROUP, p.SETMENUNAME, COALESCE(p.RECIPEID, p.INVENTORYITEMID) AS ITEMID, COALESCE(p.RECIPENAME, p.INVENTORYITEMNAME) AS ITEMNAME, CASE WHEN p.RECIPEID IS NOT NULL THEN 'RECIPE' WHEN p.INVENTORYITEMID IS NOT NULL THEN 'INVENTORY_ITEM' ELSE 'SETMENU' END AS ITEMTYPE, p.PORTIONFACTOR, p.STANDARDPORTION, p.PLANNEDPORTIONS, p.PRODUCEDPORTIONS, p.TOTALWEIGHT, p.REUSEPORTIONS, p.REUSEWEIGHT, p.WASTEPORTIONS, p.WASTEWEIGHT, p.OVERPRODUCTION, p.WEIGHTPERPORTION, p.COSTPERPORTION, p.SOLDPORTIONS FROM dbo.rvMenuPlanProductionForSetMenus AS p LEFT JOIN dbo.ARTIKEL AS art ON art.ART_ID = p.SETMENUID LEFT JOIN dbo.SETMENUGROUP AS g ON g.SMG_ID = art.SMG_NR WHERE CAST(p.MENUPLANDATE AS date) >= :dfVon AND CAST(p.MENUPLANDATE AS date) <= :dfBis ), DetailData AS ( /* ============================================================ 4. Resepti- ja artikkelitason suunnitelman ja toteuman yhdistys ============================================================ */ SELECT pl.COSTCENTERID, pl.COSTCENTERNAME, pl.MENUPLANDATE, pl.MENUPLANTEMPLATEID, pl.MENUPLANTEMPLATENAME, pl.SETMENUID, ac.SET_MENU_GROUP, pl.SETMENUNAME, pl.COS_MENU, pl.PLAN_GROUP, pl.ITEMID, pl.ITEMNAME, pl.ITEMTYPE, 'DETAIL' AS REPORT_LEVEL, COALESCE(ac.PORTIONFACTOR, pl.PORTIONFACTOR) AS PORTIONFACTOR, COALESCE(ac.STANDARDPORTION, pl.STANDARDPORTION) AS STANDARDPORTION, pl.ORIGINAL_PLANNEDPORTIONS, pl.ORIGINAL_PLANNEDWEIGHT, pl.COS_ITEM, ac.PLANNEDPORTIONS AS ACTUAL_VIEW_PLANNEDPORTIONS, ac.PRODUCEDPORTIONS, ac.TOTALWEIGHT, ac.REUSEPORTIONS, ac.REUSEWEIGHT, ac.WASTEPORTIONS, ac.WASTEWEIGHT, ac.OVERPRODUCTION, ac.WEIGHTPERPORTION, ac.COSTPERPORTION, ac.SOLDPORTIONS, /* 1. Suunniteltu kokonaispaino */ pl.ORIGINAL_PLANNEDPORTIONS * pl.ORIGINAL_PLANNEDWEIGHT AS PLANNED_TOTAL_WEIGHT, /* 2. Suunniteltu kokonaiskustannus */ pl.ORIGINAL_PLANNEDPORTIONS * pl.COS_ITEM AS PLANNED_TOTAL_COST, /* 3. Toteutunut kokonaispaino */ ac.WEIGHTPERPORTION * ac.PRODUCEDPORTIONS AS ACTUAL_TOTAL_WEIGHT, /* 4. Toteutunut kokonaiskustannus */ ac.PRODUCEDPORTIONS * ac.COSTPERPORTION AS ACTUAL_TOTAL_COST, CASE WHEN ac.ITEMID IS NULL THEN 0 ELSE 1 END AS HAS_ACTUAL_DATA, GETDATE() AS REPORTDATE FROM PlannedData AS pl LEFT JOIN ActualData AS ac ON ac.COSTCENTERID = pl.COSTCENTERID AND ac.MENUPLANDATE = pl.MENUPLANDATE AND ac.MENUPLANTEMPLATEID = pl.MENUPLANTEMPLATEID AND ac.SETMENUID = pl.SETMENUID AND ac.ITEMID = pl.ITEMID AND ac.ITEMTYPE = pl.ITEMTYPE ), SetMenuData AS ( /* ============================================================ 5. SETMENU-tason yhteenvetorivit MAX(ORIGINAL_PLANNEDPORTIONS) estää suunniteltujen annosten moninkertaistumisen reseptien lukumäärällä. Painot ja kustannukset summataan SETMENU-tasolle detail-rivien lasketuista arvoista. ============================================================ */ SELECT COSTCENTERID, COSTCENTERNAME, MENUPLANDATE, MENUPLANTEMPLATEID, MENUPLANTEMPLATENAME, SETMENUID, MAX(SET_MENU_GROUP) AS SET_MENU_GROUP, SETMENUNAME, MAX(COS_MENU) AS COS_MENU, MAX(SET_MENU_GROUP) AS PLAN_GROUP, SETMENUID AS ITEMID, SETMENUNAME AS ITEMNAME, 'SETMENU' AS ITEMTYPE, 'SETMENU' AS REPORT_LEVEL, CAST(NULL AS decimal(18,6)) AS PORTIONFACTOR, CAST(NULL AS decimal(18,6)) AS STANDARDPORTION, MAX(ORIGINAL_PLANNEDPORTIONS) AS ORIGINAL_PLANNEDPORTIONS, /* SETMENU-tason paino per suunniteltu annos. Muodostetaan resepti-/artikkelirivien painojen summana. */ SUM(COALESCE(ORIGINAL_PLANNEDWEIGHT, 0)) AS ORIGINAL_PLANNEDWEIGHT, /* SETMENU-tason kustannus per annos. Käytetään ensisijaisesti COS_MENU-arvoa. */ MAX(COS_MENU) AS COS_ITEM, MAX(ACTUAL_VIEW_PLANNEDPORTIONS) AS ACTUAL_VIEW_PLANNEDPORTIONS, /* Näissä käytetään MAX-arvoa, koska sama SETMENU-tason annosmäärä voi toistua jokaisella detail-rivillä. */ MAX(PRODUCEDPORTIONS) AS PRODUCEDPORTIONS, SUM(COALESCE(TOTALWEIGHT, 0)) AS TOTALWEIGHT, MAX(REUSEPORTIONS) AS REUSEPORTIONS, SUM(COALESCE(REUSEWEIGHT, 0)) AS REUSEWEIGHT, MAX(WASTEPORTIONS) AS WASTEPORTIONS, SUM(COALESCE(WASTEWEIGHT, 0)) AS WASTEWEIGHT, MAX(OVERPRODUCTION) AS OVERPRODUCTION, /* SETMENU-tason toteutunut paino per annos: yhteenlaskettu toteutunut paino / tuotetut annokset. */ CASE WHEN MAX(PRODUCEDPORTIONS) <> 0 THEN SUM( COALESCE( WEIGHTPERPORTION * PRODUCEDPORTIONS, 0 ) ) / MAX(PRODUCEDPORTIONS) ELSE NULL END AS WEIGHTPERPORTION, /* SETMENU-tason toteutunut kustannus per annos: yhteenlaskettu toteutunut kustannus / tuotetut annokset. */ CASE WHEN MAX(PRODUCEDPORTIONS) <> 0 THEN SUM( COALESCE( PRODUCEDPORTIONS * COSTPERPORTION, 0 ) ) / MAX(PRODUCEDPORTIONS) ELSE NULL END AS COSTPERPORTION, MAX(SOLDPORTIONS) AS SOLDPORTIONS, /* 1. SETMENU:n suunniteltu kokonaispaino */ SUM( COALESCE( ORIGINAL_PLANNEDPORTIONS * ORIGINAL_PLANNEDWEIGHT, 0 ) ) AS PLANNED_TOTAL_WEIGHT, /* 2. SETMENU:n suunniteltu kokonaiskustannus */ CASE WHEN MAX(COS_MENU) IS NOT NULL THEN MAX(ORIGINAL_PLANNEDPORTIONS) * MAX(COS_MENU) ELSE SUM( COALESCE( ORIGINAL_PLANNEDPORTIONS * COS_ITEM, 0 ) ) END AS PLANNED_TOTAL_COST, /* 3. SETMENU:n toteutunut kokonaispaino */ SUM( COALESCE( WEIGHTPERPORTION * PRODUCEDPORTIONS, 0 ) ) AS ACTUAL_TOTAL_WEIGHT, /* 4. SETMENU:n toteutunut kokonaiskustannus */ SUM( COALESCE( PRODUCEDPORTIONS * COSTPERPORTION, 0 ) ) AS ACTUAL_TOTAL_COST, MAX(HAS_ACTUAL_DATA) AS HAS_ACTUAL_DATA, MAX(REPORTDATE) AS REPORTDATE FROM DetailData GROUP BY COSTCENTERID, COSTCENTERNAME, MENUPLANDATE, MENUPLANTEMPLATEID, MENUPLANTEMPLATENAME, SETMENUID, SETMENUNAME ), FinalData AS ( /* SETMENU-yhteenvetorivit */ SELECT * FROM SetMenuData UNION ALL /* Resepti- ja artikkelitason detail-rivit */ SELECT * FROM DetailData ) SELECT COSTCENTERID, COSTCENTERNAME, MENUPLANDATE, MENUPLANTEMPLATEID, MENUPLANTEMPLATENAME, SETMENUID, SET_MENU_GROUP, SETMENUNAME, COS_MENU, PLAN_GROUP, ITEMID, ITEMNAME, ITEMTYPE, REPORT_LEVEL, PORTIONFACTOR, STANDARDPORTION, ORIGINAL_PLANNEDPORTIONS, ORIGINAL_PLANNEDWEIGHT, COS_ITEM, ACTUAL_VIEW_PLANNEDPORTIONS, PRODUCEDPORTIONS, TOTALWEIGHT, REUSEPORTIONS, REUSEWEIGHT, WASTEPORTIONS, WASTEWEIGHT, OVERPRODUCTION, WEIGHTPERPORTION, COSTPERPORTION, SOLDPORTIONS, PLANNED_TOTAL_WEIGHT, PLANNED_TOTAL_COST, ACTUAL_TOTAL_WEIGHT, ACTUAL_TOTAL_COST, HAS_ACTUAL_DATA, REPORTDATE FROM FinalData ORDER BY COSTCENTERNAME, MENUPLANDATE, MENUPLANTEMPLATENAME, SETMENUNAME, CASE WHEN REPORT_LEVEL = 'SETMENU' THEN 0 ELSE 1 END, ITEMTYPE, ITEMNAME;
